443 """Get the speed limit signs that are within the camera frame.

445 Warning: 

446 The depth frame must have a camera setup transform relative to the 

447 world.

449 Args: 

450 speed_signs (list(:py:class:`~pylot.perception.detection.speed_limit_sign.SpeedLimitSign`)): # noqa: E501 

451 List of speed limit signs in the world. 

452 depth_frame (:py:class:`~pylot.perception.depth_frame.DepthFrame`): 

453 Depth frame captured from the same position as the camera frame. 

454 segmented_frame (:py:class:`~pylot.perception.segmentation.segmented_frame.SegmentedFrame`): 

455 Segmented frame captured from the same position as the camera 

456 frame.

459 list(:py:class:`~pylot.perception.detection.speed_limit_sign.SpeedLimitSign`):

462 def match_bboxes_with_speed_signs(camera_transform, loc_bboxes, 

463 speed_signs): 

464 result = [] 

465 for location, bbox in loc_bboxes: 

466 best_ts = None 

467 best_dist = 1000000 

468 for ts in speed_signs: 

469 dist = location.distance(ts.transform.location) 

470 if dist < best_dist: 

471 best_dist = dist 

472 best_ts = ts 

473 if not best_ts: 

474 continue 

475 # Check that the sign is facing the ego vehicle. 

476 yaw_diff = (best_ts.transform.rotation.yaw - 

477 camera_transform.rotation.yaw) 

478 if yaw_diff < 0: 

479 yaw_diff += 360 

480 elif yaw_diff >= 360: 

481 yaw_diff -= 360 

482 if best_dist < 5**2 and yaw_diff > 30 and yaw_diff < 150: 

483 best_ts.bounding_box_2D = bbox 

484 result.append(best_ts) 

485 return result 

486 

487 if not isinstance(depth_frame, DepthFrame): 

488 raise ValueError( 

489 'depth_frame should be of type perception.depth_frame.DepthFrame') 

490 # Compute the 2D bounding boxes. 

491 bboxes_2d = segmented_frame.get_traffic_sign_bounding_boxes(min_width=8, 

492 min_height=9) 

493 # Transform the centers of 2D bounding boxes to 3D locations. 

494 coordinates = [bbox.get_center_point() for bbox in bboxes_2d] 

495 locations = depth_frame.get_pixel_locations(coordinates) 

496 loc_and_bboxes = zip(locations, bboxes_2d) 

497 det_speed_limits = match_bboxes_with_speed_signs( 

498 depth_frame.camera_setup.transform, loc_and_bboxes, speed_signs)
